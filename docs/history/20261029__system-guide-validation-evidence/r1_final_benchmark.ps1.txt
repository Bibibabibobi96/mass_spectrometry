param([Parameter(Mandatory)][string]$RepositoryRoot)
Set-StrictMode -Version Latest
$ErrorActionPreference='Stop'
$evidence=$PSScriptRoot
foreach($name in @('measurement.json','sample_equivalence.json','stop.txt')){
 if(Test-Path -LiteralPath (Join-Path $evidence $name)){throw "Remeasurement already has execution evidence: $name"}
}
if(@(Get-ChildItem -LiteralPath $evidence -Filter 'ready*.txt').Count){throw 'Remeasurement already has worker markers'}
$source=Join-Path $evidence 'resource_budget_support_current.ps1'
$baseline=Join-Path $evidence 'resource_budget_support_baseline.ps1'
$plan=Get-Content -LiteralPath (Join-Path $evidence 'preparation.json') -Raw|ConvertFrom-Json
foreach($item in $plan.frozen_files){
 $path=Join-Path $evidence $item.path
 if((Get-Item -LiteralPath $path).Length-ne$item.bytes-or(Get-FileHash -LiteralPath $path -Algorithm SHA256).Hash-ne$item.sha256){throw "Frozen input changed: $path"}
}
$sharedOwner=Join-Path $RepositoryRoot 'common/host_execution_lease.ps1'
$sharedPolicy=Join-Path $RepositoryRoot 'common/host_resource_policy.json'
if((Get-FileHash -LiteralPath $sharedOwner -Algorithm SHA256).Hash-ne$plan.shared_owner_sha256-or
   (Get-FileHash -LiteralPath $sharedPolicy -Algorithm SHA256).Hash-ne$plan.shared_policy_sha256){throw 'Shared admission/helper source changed after preparation'}
# Public control plane stays in the real repository; both measured arms use the same helpers.
# Load it first so the frozen sampler never derives a private TEMP resource database.
. $sharedOwner
. $source
$lease=$null
$primary=$null
$release=$null
$allStopped=$true
$workerCompletion=@()
$measurements=@()
$roots=@()
$cpuName=@(CimCmdlets\Get-CimInstance -ClassName Win32_Processor|ForEach-Object Name)
$children=@()
$stopPath=Join-Path $evidence 'stop.txt'
$started=[datetimeoffset]::UtcNow
try {
 $state=Resolve-HostResourceStatePath -StatePath $plan.central_resource_state_path
 $lease=Enter-HostResourceStage -Role GATE -Stage prepare -Budget (Get-HostResourceBudget -Role GATE -Stage prepare) -StatePath $state -RunId 'r1-frozen-source-remeasurement-20261029'
 if($lease.inherited){throw 'Run this isolated fixture from a fresh non-inherited GATE context'}
 $equivalencePath=Join-Path $evidence 'sample_equivalence.json'
 & (Join-Path $evidence 'synthetic_equivalence.ps1') -RepositoryRoot $RepositoryRoot |
  Set-Content -LiteralPath $equivalencePath -Encoding utf8
 $python=Join-Path $RepositoryRoot '.venv/Scripts/python.exe'
 & $python -X utf8 -B -c 'import json,sys; d=json.load(open(sys.argv[1],encoding="utf-8-sig")); assert len(d)==6; assert all(x["independent"]==x["shared"] for x in d); print("SAMPLE_EQUIVALENCE=PASS SCENARIOS=6")' $equivalencePath
 if($LASTEXITCODE-ne0){throw 'Complete-return dictionary equivalence failed'}
 $tokens=$null;$errors=$null
 $baseAst=[Management.Automation.Language.Parser]::ParseFile($baseline,[ref]$tokens,[ref]$errors)
 if($errors.Count){throw 'Baseline parse failed'}
 $baseFunction=$baseAst.Find({param($node) $node-is[Management.Automation.Language.FunctionDefinitionAst]-and$node.Name-eq'Get-ManagedSolverProcessSample'},$true)
 Set-Item Function:Get-BaselineManagedSolverProcessSample -Value $baseFunction.Body.GetScriptBlock()
 $originalParents=${function:Get-HostResourceProcessParents}
 function Get-Process {
  [CmdletBinding()]param([int[]]$Id)
  if($PSBoundParameters.ContainsKey('Id')){Microsoft.PowerShell.Management\Get-Process -Id $Id}
  else{$script:processCalls++;Microsoft.PowerShell.Management\Get-Process}
 }
 function Get-CimInstance {
  [CmdletBinding()]param([string]$ClassName)
  if($ClassName-eq'Win32_Process'){$script:cimCalls++}
  CimCmdlets\Get-CimInstance -ClassName $ClassName
 }
 function Get-HostResourceProcessParents {$script:parentCalls++;& $originalParents}
 $script:processCalls=0;$script:cimCalls=0;$script:parentCalls=0
 $exe=[Diagnostics.Process]::GetCurrentProcess().MainModule.FileName
 foreach($index in 1..8){
  $ready=Join-Path $evidence "ready$index.txt"
  $worker="[IO.File]::WriteAllText('$ready','ready');`$deadline=[datetime]::UtcNow.AddMinutes(5);while(-not[IO.File]::Exists('$stopPath')-and[datetime]::UtcNow-lt`$deadline){[Threading.Thread]::Sleep(50)}"
  $encoded=[Convert]::ToBase64String([Text.Encoding]::Unicode.GetBytes($worker))
  $children+=@(Start-Process -FilePath $exe -ArgumentList @('-NoProfile','-NonInteractive','-EncodedCommand',$encoded) -WindowStyle Hidden -PassThru)
 }
 $readyDeadline=[datetime]::UtcNow.AddSeconds(30)
 while(@(Get-ChildItem -LiteralPath $evidence -Filter 'ready*.txt').Count-lt8){
  if([datetime]::UtcNow-gt$readyDeadline){throw 'Inert workers did not become ready'}
  Start-Sleep -Milliseconds 50
 }
 $roots=@($children|ForEach-Object {[pscustomobject]@{id=$_.Id;ticks=$_.StartTime.ToUniversalTime().Ticks}})
 # Warm both paths once; omit these measurements from the recorded rounds.
 $warm=Get-ManagedSolverProcessSample -RootProcessIds @($roots[0].id)
 $warm=Get-BaselineManagedSolverProcessSample -RootProcessIds @($roots[0].id)
 $measurements=@()
 foreach($count in @(1,4,8)){
  foreach($round in 1..3){
   $order=if($round%2){@('BASE','NEW')}else{@('NEW','BASE')}
   foreach($variant in $order){
    $script:processCalls=0;$script:cimCalls=0;$script:parentCalls=0
    $watch=[Diagnostics.Stopwatch]::StartNew()
    $snapshot=$null
    if($variant-eq'NEW'){$snapshot=Get-RepositoryProcessSnapshot}
    foreach($root in @($roots|Select-Object -First $count)){
     $arguments=@{RootProcessIds=@($root.id);RootProcessStartedAtUtcTicks=@{([string]$root.id)=$root.ticks}}
     $sample=if($variant-eq'NEW'){Get-ManagedSolverProcessSample @arguments -ProcessSnapshot $snapshot}else{Get-BaselineManagedSolverProcessSample @arguments}
     if($sample.active_process_ids-notcontains$root.id){throw 'Inert owner disappeared'}
    }
    $watch.Stop()
    $measurements+=@{workers=$count;round=$round;variant=$variant;poll_elapsed_ms=$watch.Elapsed.TotalMilliseconds;
      get_process_full=$script:processCalls;cim_process_full=$script:cimCalls;fallback_parent_snapshots=$script:parentCalls}
   }
  }
 }
 } catch {
 $primary=$_
} finally {
 [IO.File]::WriteAllText($stopPath,'stop')
 foreach($child in $children){
  try {
   $id=$child.Id
   $child.WaitForExit()
   $workerCompletion+=@{pid=$id;exit_confirmed=$child.HasExited;returncode=$child.ExitCode}
   if(-not$child.HasExited-or$child.ExitCode-ne0){throw 'Inert worker did not exit normally'}
   $child.Dispose()
  } catch {
   $allStopped=$false
   if($null-eq$primary){$primary=$_}else{$primary.Exception.Data["worker_wait_$($child.Id)"]=$_.Exception.ToString()}
  }
 }
 if($null-ne$lease-and$allStopped){
  try {
   $release=Exit-HostResourceStage -Lease $lease -PassThru
   if($null-eq$release-or$release.status-ne'released'-or$release.token-ne$lease.token){throw 'Shared GATE release not confirmed'}
  } catch {
   if($null-eq$primary){$primary=$_}else{$primary.Exception.Data['gate_release_error']=$_.Exception.ToString()}
  }
 }
}
$report=[ordered]@{
 status=$(if($null-eq$primary){'complete'}else{'failed'})
 started_utc=$started.ToString('o');ended_utc=[datetimeoffset]::UtcNow.ToString('o')
 baseline_sha256=(Get-FileHash -LiteralPath $baseline -Algorithm SHA256).Hash
 result_sha256=(Get-FileHash -LiteralPath $source -Algorithm SHA256).Hash
 baseline_origin='Original 80292-byte mixed WORKTREE snapshot, not Git 3005980'
 source_files=@('resource_budget_support_baseline.ps1','resource_budget_support_current.ps1')
 shared_owner_sha256=$plan.shared_owner_sha256;shared_policy_sha256=$plan.shared_policy_sha256
 powershell_version=[string]$PSVersionTable.PSVersion;machine=$env:COMPUTERNAME
 logical_processors=[Environment]::ProcessorCount;cpu_name=$cpuName
 measurement_scope='Collector (NEW only) plus identical family-sampling foreach; excludes sleeps, launch, admission, dictionary fixture and output writes'
 counters_include_wrapper_dispatch=$true;warmup_pairs=1;rounds_per_worker_count=3
 workers=$roots;measurements=$measurements;worker_completion=$workerCompletion
 gate_release=$release;all_workers_natural_exit_zero=$allStopped
 whole_dispatch_executed=$false
 error=$(if($null-ne$primary){$primary.ToString()}else{$null})
}
$report|ConvertTo-Json -Depth 12|Set-Content -LiteralPath (Join-Path $evidence 'measurement.json') -Encoding utf8
Write-Output ('MEASUREMENT='+ (Join-Path $evidence 'measurement.json'))
if($null-ne$primary){throw $primary}
if($measurements.Count-ne18){throw 'Expected exactly 18 paired measurements'}
Write-Output 'R1_REMEASUREMENT=PASS'
