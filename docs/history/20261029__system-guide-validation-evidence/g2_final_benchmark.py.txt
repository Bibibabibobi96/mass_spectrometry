def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def source_entries(root, copy_to=None):
    manifest=[]
    for name in ('common','projects','integrations'):
        for directory,dirs,files in os.walk(root/name):
            dirs[:]=[d for d in dirs if d not in EXCLUDED]
            for name in files:
                path=Path(directory)/name
                if path.suffix not in SUFFIXES:
                    continue
                relative=path.relative_to(root)
                raw=path.read_bytes()
                manifest.append((relative.as_posix(),len(raw),sha(raw)))
                if copy_to is not None:
                    dest=copy_to/relative
                    dest.parent.mkdir(parents=True,exist_ok=True)
                    dest.write_bytes(raw)
    manifest.sort()
    return manifest

def manifest_digest(manifest):
    return sha(json.dumps(manifest,separators=(',',':')).encode())

def save(result):
    DEST.write_text(json.dumps(result,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
